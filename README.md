CREATE TABLE [dbo].[IDEMPOTENCY_KEY] (
    [REQUEST_SCOPE] VARCHAR (300) NOT NULL,
    [KEY_VALUE] VARCHAR (128) NOT NULL,
    [REQUEST_HASH] VARCHAR (64) NOT NULL,
    [STATUS_CODE] INT NULL,
    [CONTENT_TYPE] VARCHAR (200) NULL,
    [RESPONSE_LOCATION] VARCHAR (500) NULL,
    [RESPONSE_BODY] NVARCHAR (MAX) NULL,
    [CREATED_AT] DATETIME NOT NULL,
    [COMPLETED_AT] DATETIME NULL,
    CONSTRAINT [IDEMPOTENCY_KEY_PK] PRIMARY KEY CLUSTERED ([REQUEST_SCOPE] ASC, [KEY_VALUE] ASC)
);
